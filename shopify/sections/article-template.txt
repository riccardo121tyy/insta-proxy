{% comment %}
  Inline promo banners: split the article body on paragraph breaks and
  interleave non-modal banners promoting CAME's free tools, support chat,
  custom-service page and press-custom articles. A banner is only placed
  at a paragraph boundary where neither the paragraph just before nor the
  one right after contains an <img>, so a banner never lands next to a
  picture. Short articles stay banner-free.
{% endcomment %}

{% assign banner_every = 4 %}
{% assign banner_max = 3 %}
{% assign para_split = article.content | split: '</p>' %}
{% assign para_count = para_split.size %}

{% if para_count > 6 %}
  {% assign promo_variants = 'tools,support,custom,press' | split: ',' %}
  {% assign promo_start = article.id | modulo: 4 %}
  {% assign banners_inserted = 0 %}
  {% assign paras_since_banner = 0 %}
  {% assign article_body = '' %}

  {% for part in para_split %}
    {% assign is_last = forloop.last %}
    {% if is_last %}
      {% assign article_body = article_body | append: part %}
    {% else %}
      {% assign article_body = article_body | append: part | append: '</p>' %}
    {% endif %}

    {% assign paras_since_banner = paras_since_banner | plus: 1 %}

    {% assign this_has_img = part contains '<img' %}
    {% assign next_part = para_split[forloop.index] %}
    {% assign next_has_img = false %}
    {% if next_part %}
      {% assign next_has_img = next_part contains '<img' %}
    {% endif %}

    {% if paras_since_banner >= banner_every and this_has_img == false and next_has_img == false and is_last == false and banners_inserted < banner_max %}
      {% assign banners_inserted = banners_inserted | plus: 1 %}
      {% assign paras_since_banner = 0 %}
      {% assign variant_index = promo_start | plus: banners_inserted %}
      {% assign variant_index = variant_index | modulo: 4 %}
      {% assign variant = promo_variants[variant_index] %}

      {% case variant %}
        {% when 'tools' %}
          {% assign promo_eyebrow = 'Strumento gratuito CAME' %}
          {% if primary_tag != blank %}
            {% assign promo_title = 'Scopri i nostri strumenti gratuiti per ' | append: primary_tag %}
          {% else %}
            {% assign promo_title = 'Scopri i nostri strumenti gratuiti' %}
          {% endif %}
          {% assign promo_text = 'Nessuna registrazione richiesta, risultato immediato.' %}
          {% assign promo_cta = 'Prova ora' %}
          {% assign promo_url = 'https://www.camecompany.com/pages/tutti-i-tool-gratuiti' %}
        {% when 'support' %}
          {% assign promo_eyebrow = 'Assistenza CAME' %}
          {% assign promo_title = 'Hai un dubbio su questa guida?' %}
          {% assign promo_text = 'Scrivici in chat, rispondiamo subito e senza bot.' %}

<script>
  (function () {
    var openBtn = document.getElementById('came-author-open');
    var modal = document.getElementById('came-author-modal');
    if (!openBtn || !modal) return;

    var closers = modal.querySelectorAll('[data-came-author-close]');

    function openModal() {
      modal.classList.add('is-open');
      modal.setAttribute('aria-hidden', 'false');
      document.body.style.overflow = 'hidden';
      var closeBtn = modal.querySelector('.came-author-modal-close');
      if (closeBtn) closeBtn.focus();
    }

    function closeModal() {
      modal.classList.remove('is-open');
      modal.setAttribute('aria-hidden', 'true');
      document.body.style.overflow = '';
      openBtn.focus();
    }

    openBtn.addEventListener('click', openModal);
    closers.forEach(function (el) {
      el.addEventListener('click', closeModal);
    });

    document.addEventListener('keydown', function (e) {
      if (e.key === 'Escape' && modal.classList.contains('is-open')) closeModal();
    });
  })();
</script>
