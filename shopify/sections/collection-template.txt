<script>
  (function () {
    var root = document.currentScript.previousElementSibling;
    if (!root || !root.classList.contains('came-coll')) {
      root = document.querySelector('.came-coll');
    }
    if (!root) return;

    var input = root.querySelector('[data-came-product-search]');
    var grid = root.querySelector('[data-came-product-grid]');
    var empty = root.querySelector('[data-came-empty-search]');
    if (!input || !grid) return;

    var cards = Array.prototype.slice.call(grid.querySelectorAll('[data-came-product-search]'));

    input.addEventListener('input', function () {
      var query = input.value.trim().toLowerCase();
      var visible = 0;

      cards.forEach(function (card) {
        var text = card.getAttribute('data-came-product-search') || '';
        var match = !query || text.indexOf(query) !== -1;
        card.classList.toggle('is-hidden', !match);
        if (match) visible++;
      });

      if (empty) {
        empty.classList.toggle('is-visible', visible === 0);
      }
    });
  })();
</script>
